(() => { const WS_URL = "wss://api.derivws.com/trading/v1/options/ws/public"; const symbols = ["R_10", "R_25", "R_50", "R_75", "R_100"]; const names = { R_10: "Volatility 10", R_25: "Volatility 25", R_50: "Volatility 50", R_75: "Volatility 75", R_100: "Volatility 100" }; const histories = Object.fromEntries(symbols.map(s => [s, []])); const pipSizes = {}; const historyLoaded = {}; const liveBuffers = Object.fromEntries(symbols.map(s => [s, []])); let ws = null; let running = false; let scanAll = false; let reconnectTimer = null; const $ = id => document.getElementById(id); const selected = () => $("market").value; const MAX_TICKS = 5000; const HISTORY_COUNT = 1000; const TEST_WINDOW = 200; function setConnection(message, good = false) { $("connection").textContent = "● " + message; $("connection").style.color = good ? "#72f0a2" : "#ffb4b4"; } function extractDigit(quote, pip) { const decimals = Number.isInteger(pip) ? pip : 2; const fixed = Number(quote).toFixed(decimals); return Number(fixed.replace(/\D/g, "").slice(-1)); } function addDigit(symbol, digit) { if (!Number.isInteger(digit) || digit < 0 || digit > 9) return; histories[symbol].push(digit); if (histories[symbol].length > MAX_TICKS) { histories[symbol].shift(); } } function countsFor(data) { const counts = Array(10).fill(0); data.forEach(d => counts[d]++); return counts; } function chooseBest(counts) { let best = 0; for (let i = 1; i < 10; i++) { if (counts[i] > counts[best]) best = i; } return best; } // Rank digits using multiple historical windows. function rankDigits(data) { const recent = countsFor(data.slice(-200)); const medium = countsFor(data.slice(-1000)); const long = countsFor(data.slice(-5000)); const scores = Array(10).fill(0); for (let i = 0; i < 10; i++) { scores[i] = (recent[i] / Math.max(1, Math.min(200, data.length))) * 0.5 + (medium[i] / Math.max(1, Math.min(1000, data.length))) * 0.3 + (long[i] / Math.max(1, data.length)) * 0.2; } const ranking = scores.map((score, digit) => ({ digit, score })).sort((a, b) => b.score - a.score); return { ranking, recent, medium, long }; } // Test whether a rolling frequency strategy beat the 10% baseline. function backtest(data) { if (data.length <= TEST_WINDOW) { return { trials: 0, hits: 0, accuracy: 0 }; } let hits = 0; let trials = 0; for (let i = TEST_WINDOW; i < data.length; i++) { const previous = data.slice(i - TEST_WINDOW, i); const candidate = chooseBest(countsFor(previous)); if (data[i] === candidate) hits++; trials++; } return { trials, hits, accuracy: trials ? hits / trials * 100 : 0 }; } function renderBars(data) { const counts = countsFor(data); const total = data.length; $("bars").innerHTML = counts.map((n, i) => `
${i}
${n}
`).join(""); } function renderMarkets() { $("marketResults").innerHTML = symbols.map(symbol => { const data = histories[symbol]; const result = rankDigits(data); const test = backtest(data); const candidate = result.ranking[0]; let status = "Collecting historical data"; if (data.length > TEST_WINDOW) { status = `Candidate: ${candidate.digit} · Backtest: ${test.accuracy.toFixed(1)}% · ${test.trials} tests`; } return `
${names[symbol]}
${status}
${data.length} ticks stored
`; }).join(""); } function renderSelected() { const symbol = selected(); const data = histories[symbol]; const result = rankDigits(data); const test = backtest(data); const candidate = result.ranking[0]; $("tickCount").textContent = data.length + " ticks scanned"; $("lastDigit").textContent = data.length ? data[data.length - 1] : "—"; if (data.length < TEST_WINDOW + 1) { $("topDigit").textContent = "—"; $("signal").textContent = "COLLECTING EVIDENCE"; $("signal").style.background = "#39304f"; $("signal").style.color = "#d6c8ff"; $("explain").textContent = `Collecting data for ${names[symbol]}. At least ${TEST_WINDOW + 1} ticks are needed for backtesting.`; $("topCount").textContent = "—"; $("topRate").textContent = "—"; } else { $("topDigit").textContent = candidate.digit; $("topCount").textContent = result.recent[candidate.digit]; $("topRate").textContent = (result.recent[candidate.digit] / Math.min(200, data.length) * 100).toFixed(1) + "%"; const beatsBaseline = test.accuracy > 10; $("signal").textContent = beatsBaseline ? "ABOVE 10% BASELINE — TEST FURTHER" : "NO DEMONSTRATED ADVANTAGE"; $("signal").style.background = beatsBaseline ? "#123b2a" : "#51332b"; $("signal").style.color = beatsBaseline ? "#72f0a2" : "#ffd0a8"; $("explain").textContent = `Digit ${candidate.digit} is the current statistical candidate. ` + `A rolling historical test recorded ${test.hits} hits from ${test.trials} trials ` + `(${test.accuracy.toFixed(1)}%). Random-digit baseline: approximately 10%.`; } renderBars(data); renderMarkets(); } function requestHistory(symbol) { if (!ws || ws.readyState !== WebSocket.OPEN) return; historyLoaded[symbol] = false; liveBuffers[symbol] = []; ws.send(JSON.stringify({ ticks_history: symbol, count: HISTORY_COUNT, end: "latest", style: "ticks" })); $("marketStatus").textContent = "Loading history for " + names[symbol]; } function subscribe(symbol) { if (!ws || ws.readyState !== WebSocket.OPEN) return; ws.send(JSON.stringify({ ticks: symbol, subscribe: 1 })); } function subscribeTargets() { const targets = scanAll ? symbols : [selected()]; targets.forEach(symbol => requestHistory(symbol)); targets.forEach(symbol => subscribe(symbol)); $("marketStatus").textContent = scanAll ? "Scanning all five markets" : "Scanning " + names[selected()]; } function connect() { if (ws && ( ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING )) return; clearTimeout(reconnectTimer); setConnection("Connecting…"); try { ws = new WebSocket(WS_URL); } catch (error) { setConnection("Connection error"); return; } ws.onopen = () => { setConnection("Connected", true); subscribeTargets(); }; ws.onmessage = event => { let msg; try { msg = JSON.parse(event.data); } catch (error) { return; } if (msg.error) { $("marketStatus").textContent = msg.error.message || "Market feed error"; return; } // Historical tick response. if (msg.msg_type === "history" && msg.history) { const symbol = msg.echo_req?.ticks_history; if (!symbols.includes(symbol)) return; const pip = Number.isInteger(msg.pip_size) ? msg.pip_size : 2; pipSizes[symbol] = pip; const prices = msg.history.prices || []; const historicalDigits = prices.map(price => extractDigit(price, pip) ); histories[symbol] = historicalDigits.slice(-MAX_TICKS); // Add any live ticks received while history was loading. liveBuffers[symbol].forEach(digit => addDigit(symbol, digit)); liveBuffers[symbol] = []; historyLoaded[symbol] = true; $("marketStatus").textContent = "History loaded: " + names[symbol]; renderSelected(); return; // Live tick response. if (msg.msg_type === "tick" && msg.tick) { const tick = msg.tick; const symbol = tick.symbol; if (!symbols.includes(symbol)) return; if (tick.quote === undefined) return; const pip = Number.isInteger(tick.pip_size) ?
